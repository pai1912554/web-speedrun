LESSON 03

JavaScript 6 อย่างที่ตอบโจทย์แข่งได้ 90%

ไม่ต้องเรียน closure, prototype, async generator. ตอนแข่งคุณใช้แค่ 6 ท่านี้ วนไปวนมา.

ก่อนเริ่ม: เปิด DevTools ด้วย F12 แท็บ Console ค้างไว้ตลอดเวลาเขียนโค้ด. JS พังจะขึ้นตัวแดงที่นี่. ไม่เปิด = เดาไปเรื่อยๆ เสียเวลาเป็นชั่วโมง.

ท่า 1 — จับ element

const el   = document.querySelector('#total');    // ตัวแรกที่เจอ
const list = document.querySelectorAll('.card');  // ได้หลายตัว

el.textContent = 'สวัสดี';        // เปลี่ยนข้อความ
el.innerHTML   = '<b>ตัวหนา</b>'; // ใส่ HTML
el.value                          // อ่านค่าจาก input
el.classList.add('active');
el.classList.remove('active');
el.classList.toggle('active');    // มีก็เอาออก ไม่มีก็ใส่ <-- ใช้บ่อยสุด
el.style.display = 'none';

querySelector ใช้ selector แบบเดียวกับ CSS เป๊ะ: #id, .class, tag. จำอันเดียวพอ.

ท่า 2 — ดักการกด (event)

document.querySelector('#btn').addEventListener('click', () => {
  console.log('กดแล้ว');
});

input.addEventListener('input',  e => console.log(e.target.value)); // พิมพ์ทีละตัว
select.addEventListener('change', e => console.log(e.target.value));
form.addEventListener('submit', e => { e.preventDefault(); /* สำคัญ! */ });
กับดัก: ลืม e.preventDefault() ในฟอร์ม ทำให้หน้าเว็บรีเฟรช ข้อมูลหายหมด. มือใหม่ตายตรงนี้ทุกคน.

ท่า 3 — ข้อมูล + render list (ท่าที่ใช้มากที่สุดในการแข่ง)

โจทย์แข่ง 80% คือ "แสดงรายการของ + ค้นหา/กรอง/เพิ่ม/ลบ". สูตรตายตัว:

// 1) ข้อมูล -- array ของ object เสมอ
let items = [
  { id:1, name:'กาแฟ',  price:45, cat:'ดื่ม' },
  { id:2, name:'ขนมปัง', price:30, cat:'กิน' },
];

// 2) ที่ว่างใน HTML:  <div id="list"></div>
const box = document.querySelector('#list');

// 3) ฟังก์ชัน render -- เรียกทุกครั้งที่ข้อมูลเปลี่ยน
function render(data = items){
  box.innerHTML = data.map(it => `
    <div class="card">
      <h3>${it.name}</h3>
      <p>${it.price} บาท · ${it.cat}</p>
      <button onclick="del(${it.id})">ลบ</button>
    </div>`).join('');
}
render();

เข้าใจบรรทัดนี้ = เข้าใจ 80% ของ front-end: data.map(...).join('') คือ "แปลงของทุกชิ้นเป็น HTML แล้วต่อกันเป็นข้อความเดียว".

เพิ่ม / ลบ / กรอง — ทุกอันจบด้วย render():

function add(name, price){
  items.push({ id: Date.now(), name, price, cat:'ใหม่' });
  render();
}
function del(id){
  items = items.filter(it => it.id !== id);
  render();
}
// ค้นหา
search.addEventListener('input', e => {
  const q = e.target.value.toLowerCase();
  render( items.filter(it => it.name.toLowerCase().includes(q)) );
});
// เรียงราคา
items.sort((a,b) => a.price - b.price);   // น้อยไปมาก
items.sort((a,b) => b.price - a.price);   // มากไปน้อย
// รวมเงิน
const total = items.reduce((sum, it) => sum + it.price, 0);
เมธอดทำอะไรโจทย์ที่เจอ
.map()แปลงทุกตัวสร้าง HTML การ์ด/แถวตาราง
.filter()คัดเฉพาะที่ผ่านค้นหา, กรองหมวด, ลบ
.reduce()ยุบเหลือค่าเดียวรวมยอดเงิน, นับ
.sort()เรียงเรียงราคา/ชื่อ/วันที่
.find()หาตัวแรกที่ตรงหา item จาก id
.forEach()วนทำอะไรสักอย่างผูก event หลายปุ่ม

ท่า 4 — ฟอร์ม + ตรวจข้อมูล

form.addEventListener('submit', e => {
  e.preventDefault();
  const name  = form.name.value.trim();   // form.[ค่า attribute name]
  const email = form.email.value.trim();

  if(!name){ alert('กรอกชื่อด้วย'); return; }
  if(!email.includes('@')){ alert('อีเมลไม่ถูก'); return; }

  console.log({name, email});
  form.reset();
});

HTML ช่วยตรวจให้ฟรี ไม่ต้องเขียน JS: required, type="email", min="0", maxlength="20", pattern.

ท่า 5 — ดึงข้อมูลจาก API

async function load(){
  try{
    const res  = await fetch('https://jsonplaceholder.typicode.com/users');
    const data = await res.json();
    items = data;
    render();
  }catch(err){
    box.innerHTML = '<p>โหลดข้อมูลไม่สำเร็จ</p>';
    console.error(err);
  }
}
load();
กับดักตอนแข่ง: fetch ไม่ทำงานเมื่อเปิดไฟล์แบบ file:/// (ติด CORS). ต้องเปิดผ่าน Live Server เท่านั้น. ถ้า API ล่ม/เน็ตไม่มี ให้ใช้ข้อมูลปลอมใน array ทันที อย่าดื้อ.

API ฟรีไว้เดโม (ไม่ต้องสมัคร): jsonplaceholder.typicode.com, fakestoreapi.com (สินค้า+รูป), dog.ceo/api/breeds/image/random, open-meteo.com (อากาศ), picsum.photos/400/300 (รูปสุ่ม).

ท่า 6 — เก็บข้อมูลไม่ให้หายตอนรีเฟรช

localStorage.setItem('items', JSON.stringify(items));       // เซฟ
items = JSON.parse(localStorage.getItem('items')) || [];    // โหลด (|| [] กันค่าว่าง)
localStorage.removeItem('items');

ใส่ save() ต่อท้ายทุกจุดที่แก้ข้อมูล. กรรมการรีเฟรชแล้วข้อมูลยังอยู่ = ดูเป็นระบบจริง ได้คะแนนง่ายมาก.

เบ็ดเตล็ดที่ใช้บ่อย

Number(x)                            // ข้อความเป็นตัวเลข ("5"+1="51" แต่ Number("5")+1=6)
n.toLocaleString('th-TH')            // 1234567 เป็น "1,234,567"
new Date().toLocaleDateString('th-TH')
Math.round(x)  Math.random()
setTimeout(fn, 1000)   setInterval(fn, 1000)
el.scrollIntoView({behavior:'smooth'})
